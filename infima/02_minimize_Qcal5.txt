############################################################
# Two-variable check of the lower bound Qcal5star(alpha,beta).
# Qcal5(alpha,beta,zeta) >= Qcal5star(alpha,beta) for 0<=zeta<=81.
#
# All numerical Meijer evaluations are at zeta=81; the needed
# zeta=0 endpoint is evaluated using its explicit gamma value.
#
# Expected boundary infimum: 3.12745586485783363468...,
# approached at (alpha,beta)=(0,1).
#
# Run this entire standalone file. It scans a finite parameter
# grid and reports every failed evaluation; it does not certify
# a continuous global minimum by itself.
############################################################

restart:
Digits := 60:

# Interior, endpoint-biased parameter grid. Exact endpoint limits are
# evaluated separately; the origin has directional limits.
Nalpha := 61:
Nbeta := 61:
eps := 10^(-10):
Kkeep := 10:

CosGrid := proc(L,R,N)
    local i;
    if N <= 1 then return [evalf(L)]; end if;
    return [seq(evalf(L+(R-L)*(1-cos(Pi*i/(N-1)))/2),
                i=0..N-1)];
end proc:

alphaGrid := CosGrid(eps,1-eps,Nalpha):
betaGrid := CosGrid(eps,1-eps,Nbeta):

IsFiniteReal := proc(v)
    return evalb(type(v,realcons)
                 and not has(v,undefined) and not has(v,infinity));
end proc:

# Meijer integrals at zeta=81 only, for k=0 or k=1.
# Their zeta=0 values, when needed below, are explicit gamma values.
M1_81 := proc(a::numeric,b::numeric,k::nonnegint)
    return evalf(
        81^((1+k+a-b)/2)/(2*sqrt(Pi)*GAMMA(-a/2))
        *MeijerG([[(1-k+b)/2],[]],
                 [[-(1+k+a-b)/2,0,1/2],[]],81/4)
    );
end proc:

M2_81 := proc(a::numeric,b::numeric,k::nonnegint)
    return evalf(
        81^((1+k+a-b)/2)/(2*sqrt(Pi)*GAMMA(b/2))
        *MeijerG([[(1-k-a)/2],[]],
                 [[-(1+k+a-b)/2,0,1/2],[]],81/4)
    );
end proc:

InsertBest := proc(records,item,K)
    local L;
    L := sort([op(records),item],(x,y)->evalb(x[1]<y[1]));
    if nops(L)>K then return L[1..K]; end if;
    return L;
end proc:

# Interior evaluator of the piecewise lower bound.
Qcal5star := proc(a::numeric,b::numeric)
    local aa,bb,psi0,main,penalty;
    aa := evalf(a);
    bb := evalf(b);

    psi0 := sin(Pi*bb)*M1_81(aa,bb,0)
            +sin(Pi*aa)*M2_81(aa,bb,0);

    main := Pi*10^aa/GAMMA(bb)
        +sin(Pi*aa)*GAMMA(1+aa)*(1-aa-bb)*10^(-bb)
        -(1-aa)*psi0;

    penalty := 0;
    if aa < bb then
        # For alpha<beta, M21(alpha,beta,0) = Gamma(2+alpha-beta).
        penalty := aa*(bb-aa)/10
            *(sin(Pi*bb)*M1_81(aa,bb,1)
              +sin(Pi*aa)*GAMMA(2+aa-bb));
    end if;

    return evalf((main-penalty)/(aa*bb*(aa+bb)));
end proc:

# Exact limiting expression for the expected corner value.
referenceValue := evalf(Pi*(1+ln(10/9)
    -MeijerG([[1/2],[]],[[0,0,1/2],[]],81/4)/(2*Pi))):
printf("Boundary reference Qcal5star: %.20g\n",referenceValue):
printf("Reference point: alpha tends to 0; beta tends to 1.\n"):

# Main two-dimensional scan. Records are [value,alpha,beta].
bestVal := infinity:
bestRec := []:
bestList := []:
nSuccess := 0:
nFailed := 0:
failedCases := []:

printf("Scanning %d (alpha,beta) pairs at %d-digit precision.\n",
       nops(alphaGrid)*nops(betaGrid),Digits):

for a in alphaGrid do
    for b in betaGrid do
        try
            v := evalf(Qcal5star(a,b)):
        catch:
            v := Float(undefined):
        end try:

        if IsFiniteReal(v) then
            nSuccess := nSuccess+1:
            bestList := InsertBest(bestList,[v,a,b],Kkeep):
            if v < bestVal then
                bestVal := v:
                bestRec := [v,a,b]:
                printf("NEW BEST %.16g at (alpha,beta)=(%.12g, %.12g)\n",
                       v,a,b):
            end if:
        else
            nFailed := nFailed+1:
            if nops(failedCases)<10 then
                failedCases := [op(failedCases),[a,b]]:
            end if:
        end if:
    end do:
end do:

printf("\nFinite evaluations: %d; failed evaluations: %d\n",
       nSuccess,nFailed):

if nSuccess = 0 then
    printf("No finite values were obtained.\n"):
else
    printf("Smallest sampled value: %.20g\n",bestRec[1]):
    printf("At (alpha,beta)=(%.16g, %.16g)\n",bestRec[2],bestRec[3]):
    printf("Sampled value minus boundary reference: %.8e\n",
           bestRec[1]-referenceValue):
    printf("Top-%d sampled candidates:\n",nops(bestList)):
    for i from 1 to nops(bestList) do
        printf("  %d: %.16g at (%.12g, %.12g)\n",
               i,bestList[i][1],bestList[i][2],bestList[i][3]):
    end do:
end if:

if nFailed > 0 then
    printf("Inspect failed cases before accepting this run. First cases:\n"):
    print(failedCases):
end if:
