############################################################
# Two-variable check of the lower bound Hcal5star(alpha,beta).
# Hcal5(alpha,beta,zeta) >= Hcal5star(alpha,beta) for 0<=zeta<=81.
#
# All numerical Meijer evaluations are at zeta=81. The zeta=0
# endpoint in the alpha<beta branch is an explicit gamma value.
# 
# Expected boundary infimum: 0.03881373940680938519...,
# approached at alpha=0, beta=0.3390014599579164....
# 
# Run this entire standalone file. It scans a finite parameter
# grid, then refines in beta at alpha=1e-12, 1e-16 and 1e-20.
# Failed evaluations are reported separately for both stages.
# These numerical searches do not certify a continuous global minimum.
############################################################

restart:
Digits := 80:

# Interior, endpoint-biased parameter grid. Exact endpoint limits are
# evaluated separately; the origin has directional limits.
Nalpha := 121:
Nbeta := 241:
eps := 10^(-12):
Kkeep := 10:

# Local beta minimizations after the grid scan. Golden-section
# refinement uses values only, with no Meijer parameter derivatives.
refineAlpha := [eps,eps/10^4,eps/10^8]:
refineDigits := 120:
betaTol := 10^(-24):
maxRefineSteps := 200:

CosGrid := proc(L,R,N)
    local i;
    if N <= 1 then return [evalf(L)]; end if;
    return [seq(evalf(L+(R-L)*(1-cos(Pi*i/(N-1)))/2),
                i=0..N-1)];
end proc:

alphaGrid := CosGrid(eps,1-eps,Nalpha):
betaGrid := CosGrid(eps,1-eps,Nbeta):

IsFiniteReal := proc(v)
    return evalb(type(v,realcons)
                 and not has(v,undefined) and not has(v,infinity));
end proc:

# Meijer integrals at zeta=81 only, for k=0 or k=1.
# Their zeta=0 values, when needed below, are explicit gamma values.
M1_81 := proc(a::numeric,b::numeric,k::nonnegint)
    return evalf(
        81^((1+k+a-b)/2)/(2*sqrt(Pi)*GAMMA(-a/2))
        *MeijerG([[(1-k+b)/2],[]],
                 [[-(1+k+a-b)/2,0,1/2],[]],81/4)
    );
end proc:

M2_81 := proc(a::numeric,b::numeric,k::nonnegint)
    return evalf(
        81^((1+k+a-b)/2)/(2*sqrt(Pi)*GAMMA(b/2))
        *MeijerG([[(1-k-a)/2],[]],
                 [[-(1+k+a-b)/2,0,1/2],[]],81/4)
    );
end proc:

InsertBest := proc(records,item,K)
    local L;
    L := sort([op(records),item],(x,y)->evalb(x[1]<y[1]));
    if nops(L)>K then return L[1..K]; end if;
    return L;
end proc:

# Interior evaluator of the piecewise lower bound.
Hcal5star := proc(a::numeric,b::numeric)
    local aa,bb,m10,m20,m21,psi0,psi1,C5,main,correction;
    aa := evalf(a);
    bb := evalf(b);

    m10 := M1_81(aa,bb,0);
    m20 := M2_81(aa,bb,0);
    m21 := M2_81(aa,bb,1);
    psi0 := sin(Pi*bb)*m10+sin(Pi*aa)*m20;

    # sin(Pi*beta)*Gamma(1-beta) = Pi/Gamma(beta).
    C5 := Pi*2^aa/GAMMA(bb)*5^(bb-1)
            *(1+(aa-bb)*(1-bb)/10)
          +sin(Pi*aa)*2^(-bb)*GAMMA(1+aa)*5^(-1-aa)
            *(1+(aa-bb)*(1+aa)/10);

    main := C5-5^(-1-aa+bb)*psi0;
    if aa >= bb then
        # For alpha>=beta, use Psi1(alpha,beta,81).
        psi1 := sin(Pi*bb)*M1_81(aa,bb,1)+sin(Pi*aa)*m21;
        main := main-(aa-bb)/2*5^(-2-aa+bb)*psi1;
    else
        # For alpha<beta, M11(alpha,beta,0) = Gamma(2+alpha-beta).
        psi1 := sin(Pi*bb)*GAMMA(2+aa-bb)+sin(Pi*aa)*m21;
        main := main+(bb-aa)/2*5^(-2-aa+bb)*psi1;
    end if;

    # The normalized sigma correction; common factors are cancelled.
    correction := (127/5000)*GAMMA(2-bb)*4^(-2+bb)
        *(sin(Pi*aa)+sin(Pi*bb))/(aa*(1-aa)+bb*(1-bb));

    return evalf(main/(aa*bb*(aa+bb))-correction);
end proc:

# Local beta minimization on an interval selected by the grid scan.
# Returns [value,beta,final interval width,iteration count].
MinimizeBeta := proc(aa::numeric,bL::numeric,bR::numeric,tol::numeric)
    local L,R,g,c,d,fc,fd,mid,fmid,steps;
    L := evalf(bL);
    R := evalf(bR);
    g := evalf((sqrt(5)-1)/2);
    c := R-g*(R-L);
    d := L+g*(R-L);
    fc := Hcal5star(aa,c);
    fd := Hcal5star(aa,d);
    if not IsFiniteReal(fc) or not IsFiniteReal(fd) then
        error "nonfinite initial refinement value";
    end if;

    steps := 0;
    while R-L>tol and steps<maxRefineSteps do
        if fc<fd then
            R := d;
            d := c;
            fd := fc;
            c := R-g*(R-L);
            fc := Hcal5star(aa,c);
        else
            L := c;
            c := d;
            fc := fd;
            d := L+g*(R-L);
            fd := Hcal5star(aa,d);
        end if;
        if not IsFiniteReal(fc) or not IsFiniteReal(fd) then
            error "nonfinite refinement value";
        end if;
        steps := steps+1;
    end do;
    if R-L>tol then error "refinement iteration limit reached"; end if;
    mid := (L+R)/2;
    fmid := Hcal5star(aa,mid);
    if not IsFiniteReal(fmid) then error "nonfinite final value"; end if;
    if fc<fmid then mid := c; fmid := fc; end if;
    if fd<fmid then mid := d; fmid := fd; end if;
    return [evalf(fmid),evalf(mid),evalf(R-L),steps];
end proc:

referenceValue := 0.03881373940680938519125115754:
printf("Boundary reference Hcal5star: %.20g\n",referenceValue):
printf("Reference beta: 0.3390014599579164...; alpha tends to 0.\n"):

# Main two-dimensional scan. Records are [value,alpha,beta].
bestVal := infinity:
bestRec := []:
bestList := []:
nSuccess := 0:
nFailed := 0:
failedCases := []:
edgeBest := []:

printf("Scanning %d (alpha,beta) pairs at %d-digit precision.\n",
       nops(alphaGrid)*nops(betaGrid),Digits):

for a in alphaGrid do
    for b in betaGrid do
        try
            v := evalf(Hcal5star(a,b)):
        catch:
            v := Float(undefined):
        end try:

        if IsFiniteReal(v) then
            nSuccess := nSuccess+1:
            bestList := InsertBest(bestList,[v,a,b],Kkeep):
            if a=alphaGrid[1] then
                if nops(edgeBest)=0 then
                    edgeBest := [v,a,b]:
                elif v<edgeBest[1] then
                    edgeBest := [v,a,b]:
                end if:
            end if:
            if v < bestVal then
                bestVal := v:
                bestRec := [v,a,b]:
                printf("NEW BEST %.16g at (alpha,beta)=(%.12g, %.12g)\n",
                       v,a,b):
            end if:
        else
            nFailed := nFailed+1:
            if nops(failedCases)<10 then
                failedCases := [op(failedCases),[a,b]]:
            end if:
        end if:
    end do:
end do:

printf("\nFinite evaluations: %d; failed evaluations: %d\n",
       nSuccess,nFailed):

if nSuccess = 0 then
    printf("No finite values were obtained.\n"):
else
    printf("Smallest sampled value: %.20g\n",bestRec[1]):
    printf("At (alpha,beta)=(%.16g, %.16g)\n",bestRec[2],bestRec[3]):
    printf("Sampled value minus boundary reference: %.8e\n",
           bestRec[1]-referenceValue):
    printf("Top-%d sampled candidates:\n",nops(bestList)):
    for i from 1 to nops(bestList) do
        printf("  %d: %.16g at (%.12g, %.12g)\n",
               i,bestList[i][1],bestList[i][2],bestList[i][3]):
    end do:
end if:

# Select the two neighbouring beta grid points around the lowest
# value on the smallest-alpha row, then refine inside that interval.
if nops(edgeBest)>0 then
    iBest := 1:
    for j from 2 to nops(betaGrid) do
        if abs(betaGrid[j]-edgeBest[3])
           <abs(betaGrid[iBest]-edgeBest[3]) then iBest := j: end if:
    end do:
    betaLeft := betaGrid[max(1,iBest-1)]:
    betaRight := betaGrid[min(nops(betaGrid),iBest+1)]:
    refineResults := []:
    nRefineFailed := 0:
    gridDigits := Digits:
    Digits := refineDigits:
    printf("\nSmall-alpha beta minimizations on [%.16g, %.16g]\n",
           betaLeft,betaRight):
    printf("Refinement precision: %d digits.\n",Digits):
    for aSmall in map(evalf,refineAlpha) do
        try
            rec := MinimizeBeta(aSmall,betaLeft,betaRight,betaTol):
            refineResults := [op(refineResults),[rec[1],aSmall,rec[2]]]:
            printf("alpha=%.8e: beta=%.25g, Hcal5star=%.25g\n",
                   aSmall,rec[2],rec[1]):
            printf("  minus stored reference: %.10e; interval width: %.4e\n",
                   rec[1]-referenceValue,rec[3]):
        catch:
            nRefineFailed := nRefineFailed+1:
            printf("Refinement failed at alpha=%.8e\n",aSmall):
        end try:
    end do:
    printf("Successful refinements: %d; failed refinements: %d\n",
           nops(refineResults),nRefineFailed):
    Digits := gridDigits:
else
    printf("No finite smallest-alpha row was available for refinement.\n"):
end if:

if nFailed > 0 then
    printf("Inspect failed cases before accepting this run. First cases:\n"):
    print(failedCases):
end if:
