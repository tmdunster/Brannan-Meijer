############################################################
# Endpoint check away from phi=0 using c=s/phi coordinates
#
# This uses the same alpha=beta=0 expression as file 08, on a
# broader supplemental c-rectangle: phiL<=phi<=phi0 and cL<=c<=cR.
# Here cL=eps/phi0 and cR=5/phiL, and s=c*phi. The rectangle
# contains the file-08 search domain but also additional points.
############################################################

restart:
Digits := 50:

with(Optimization):

phi0 := 0.061:
phiL := 0.03:
eps  := 1e-10:

cL := eps/phi0:
cR := 5/phiL:

Lraw00_c :=
1/(2*Pi)*
(
  2*ln(sqrt((exp(c*phi)-1)^2+4*sin(phi/2)^2*exp(c*phi))/(exp(c*phi)+1))*ln(exp(c*phi)-1)
  + ln(exp(c*phi)+1)^2
  - ln(sqrt((exp(c*phi)-1)^2+4*sin(phi/2)^2*exp(c*phi)))^2
  + ln(sin(phi/2))*ln(4*sin(phi/2)/(c*phi)^2)
):

L0small00_c :=
exp(-c*phi)*(c*phi)^(-1/2)*sqrt(phi)*Lraw00_c:

ans := Minimize(
  L0small00_c,
  phi=phiL..phi0,
  c=cL..cR,
  initialpoint={phi=phiL, c=0.53}
):

printf("minimum = %.10e\n", ans[1]);
printf("phi     = %.10e\n", eval(phi,ans[2]));
printf("c       = %.10e\n", eval(c,ans[2]));
printf("s       = %.10e\n", eval(phi,ans[2])*eval(c,ans[2]));
